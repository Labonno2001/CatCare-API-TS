// Validate partial cat profile updates
const isValidCatProfileInput = (
    body: unknown
): body is UpdateCatProfileInput => {
    if (
        typeof body !== "object" ||
        body === null ||
        Array.isArray(body)
    ) {
        return false;
    }

    const input = body as Record<string, unknown>;

    const allowedFields = [
        "name",
        "age",
        "weightKg",
        "favoriteFood",
        "activity"
    ];

    const keys = Object.keys(input);

    if (
        keys.length === 0 ||
        keys.some((key) => !allowedFields.includes(key))
    ) {
        return false;
    }

    if (
        input.name !== undefined &&
        (typeof input.name !== "string" || !input.name.trim())
    ) {
        return false;
    }

    if (
        input.age !== undefined &&
        (
            typeof input.age !== "number" ||
            !Number.isFinite(input.age) ||
            input.age < 0
        )
    ) {
        return false;
    }

    if (
        input.weightKg !== undefined &&
        (
            typeof input.weightKg !== "number" ||
            !Number.isFinite(input.weightKg) ||
            input.weightKg <= 0
        )
    ) {
        return false;
    }

    if (
        input.favoriteFood !== undefined &&
        (
            typeof input.favoriteFood !== "string" ||
            !input.favoriteFood.trim()
        )
    ) {
        return false;
    }

    if (
        input.activity !== undefined &&
        (
            typeof input.activity !== "string" ||
            !input.activity.trim()
        )
    ) {
        return false;
    }

    return true;
};

import { Request, Response } from "express";

// Validate YYYY-MM-DD format and actual calendar date
const isValidDate = (date: string): boolean => {
    if (!/^\d{4}-\d{2}-\d{2}$/.test(date)) {
        return false;
    }

    const parsedDate = new Date(`${date}T00:00:00.000Z`);

    return (
        !Number.isNaN(parsedDate.getTime()) &&
        parsedDate.toISOString().slice(0, 10) === date
    );
};

// GET daily wellness information
export const getDailyWellnessController = (
    req: Request,
    res: Response
): void => {
    const date = req.query.date;

    if (typeof date !== "string" || !isValidDate(date)) {
        res.status(400).json({
            success: false,
            message: "Please provide a valid date in YYYY-MM-DD format."
        });
        return;
    }

    const wellness = getDailyWellness(date);

    if (!wellness) {
        res.status(404).json({
            success: false,
            message: "Cat profile not found."
        });
        return;
    }

    res.status(200).json({
        success: true,
        ...wellness
    });
};

import db from "../database/database";
import { CatProfile } from "../types/cat.types";

export interface UpdateCatProfileInput {
    name?: string;
    age?: number;
    weightKg?: number;
    favoriteFood?: string;
    activity?: string;
}

interface CatDatabaseRow {
    id: number;
    name: string;
    age: number;
    weight_kg: number;
    favorite_food: string;
    activity: string;
}

// Get the cat profile
export const getCatProfile = (): CatProfile | null => {
    const row = db
        .prepare("SELECT * FROM cat_profile WHERE id = ?")
        .get(1) as CatDatabaseRow | undefined;

    if (!row) return null;

    return {
        id: row.id,
        name: row.name,
        age: row.age,
        weightKg: row.weight_kg,
        favoriteFood: row.favorite_food,
        activity: row.activity
    };
};

// Update selected cat profile fields
export const updateCatProfile = (
    input: UpdateCatProfileInput
): CatProfile | null => {
    const updates: string[] = [];
    const values: (string | number)[] = [];

    if (input.name !== undefined) {
        updates.push("name = ?");
        values.push(input.name);
    }

    if (input.age !== undefined) {
        updates.push("age = ?");
        values.push(input.age);
    }

    if (input.weightKg !== undefined) {
        updates.push("weight_kg = ?");
        values.push(input.weightKg);
    }

    if (input.favoriteFood !== undefined) {
        updates.push("favorite_food = ?");
        values.push(input.favoriteFood);
    }

    if (input.activity !== undefined) {
        updates.push("activity = ?");
        values.push(input.activity);
    }

    if (updates.length === 0) {
        return getCatProfile();
    }

    values.push(1);

    db.prepare(`
        UPDATE cat_profile
        SET ${updates.join(", ")}
        WHERE id = ?
    `).run(...values);

    return getCatProfile();
};

import db from "../database/database";

import {

    FeedingDatabaseRow,
    CreateFeedingInput

// Convert SQLite row into API response format
const formatFeeding = (
    row: FeedingDatabaseRow
): FeedingRecord => {
    return {
        id: row.id,
        foodName: row.food_name,
        quantityGrams: row.quantity_grams,
        mealTime: row.meal_time,
        notes: row.notes,
        createdAt: row.created_at
    };
};

// 1. Get all feeding records
export const getAllFeedingRecords = (): FeedingRecord[] => {
    const rows = db
        .prepare("SELECT * FROM feeding_records ORDER BY id ASC")
        .all() as FeedingDatabaseRow[];

    return rows.map(formatFeeding);
};

// 2. Create a feeding record
export const createFeedingRecord = (
    input: CreateFeedingInput
): FeedingRecord => {
    const result = db.prepare(`
        INSERT INTO feeding_records
            (food_name, quantity_grams, meal_time, notes, created_at)
        VALUES (?, ?, ?, ?, ?)
    `).run(
        input.foodName.trim(),
        input.quantityGrams,
        input.mealTime,
        input.notes ?? "",
        new Date().toISOString()
    );

    const row = db.prepare(
        "SELECT * FROM feeding_records WHERE id = ?"
    ).get(result.lastInsertRowid) as FeedingDatabaseRow;

    return formatFeeding(row);

async function apiRequest(endpoint, options = {}) {
    const response = await fetch(`${API_URL}${endpoint}`, {
        ...options,
        headers: {
            ...(options.body ? { "Content-Type": "application/json" } : {}),
            ...options.headers
        }
    });

    const result = await response.json().catch(() => ({}));

    if (!response.ok || result.success === false) {
        throw new Error(result.message || `Request failed: ${response.status}`);
    }

    return result;
}

function showMessage(message, type = "success") {
    const element = $("message");

    if (!element) {
        alert(message);
        return;
    }

    element.textContent = message;
    element.className = `message ${type}`;
    element.style.display = "block";

    window.setTimeout(() => {
        element.style.display = "none";
    }, 4000);
}

        refreshApp();

    if (pageName === "profile") {
        loadCat();

    if (pageName === "feeding") {
        loadFeeding();
    }

    if (pageName === "water") {
        loadWater();

    if (pageName === "wellness") {
        loadWellness();

async function loadWater() {
    try {
        const result = await apiRequest("/api/water");

        waterRecords = result.records || result.data || [];

        renderWater();
        updateDashboard();

        return waterRecords;
    } catch (error) {
        showMessage(error.message, "error");
        throw error;
    }
}


function renderWater() {
    const table = $("water-table");

    if (!table) return;

    table.innerHTML = "";

    if (waterRecords.length === 0) {
        table.innerHTML = `
            <tr>
                <td colspan="5">No water records found.</td>
            </tr>
        `;
        return;
    }

    waterRecords.forEach(record => {
        const row = document.createElement("tr");

        const values = [
            `${record.offeredMl} ml`,
            record.measured ? `${record.consumedMl} ml` : "Not measured",
            record.measured ? "Yes" : "No",
            record.notes || ""
        ];

        values.forEach(value => {
            const cell = document.createElement("td");
            cell.textContent = value ?? "";
            row.appendChild(cell);
        });

        const actions = document.createElement("td");

        const editButton = document.createElement("button");
        editButton.textContent = "Edit";
        editButton.type = "button";
        editButton.addEventListener("click", () => editWater(record.id));

        const deleteButton = document.createElement("button");
        deleteButton.textContent = "Delete";
        deleteButton.type = "button";
        deleteButton.addEventListener("click", () => deleteWater(record.id));

        actions.append(editButton, deleteButton);
        row.appendChild(actions);

        table.appendChild(row);
    });

    const totalOffered = waterRecords.reduce(
        (sum, record) => sum + Number(record.offeredMl || 0), 0
    );

    const totalConsumed = waterRecords.reduce(
        (sum, record) =>
            sum + (record.measured ? Number(record.consumedMl || 0) : 0),
        0
    );

    if ($("water-total-offered")) {
        $("water-total-offered").textContent = `${totalOffered} ml`;
    }

    if ($("water-total-consumed")) {
        $("water-total-consumed").textContent = `${totalConsumed} ml`;
    }
}

// Edit water record
async function editWater(id) {
    const record = waterRecords.find(item => item.id === id);

    if (!record) return;

    const offered = prompt("Water offered (ml):", record.offeredMl);
    if (offered === null) return;

    const measured = confirm("Do you have a measured consumption value?");

    let consumed = null;

    if (measured) {
        consumed = prompt(
            "Water consumed (ml):",
            record.consumedMl ?? ""
        );

        if (consumed === null) return;

        consumed = Number(consumed);
    }

    const notes = prompt("Notes:", record.notes || "");
    if (notes === null) return;

    const input = {
        offeredMl: Number(offered),
        measured,
        consumedMl: measured ? consumed : null,
        notes: notes.trim()
    };

    if (
        !Number.isFinite(input.offeredMl) ||
        input.offeredMl < 0 ||
        (measured && (!Number.isFinite(consumed) || consumed < 0))
    ) {
        showMessage("Invalid water information.", "error");
        return;
    }

    try {
        await apiRequest(`/api/water/${id}`, {
            method: "PATCH",
            body: JSON.stringify(input)
        });

        await loadWater();

        showMessage("Water record updated!");
    } catch (error) {
        showMessage(error.message, "error");
    }
}

function isToday(dateValue) {
    if (!dateValue) return false;

    const date = new Date(dateValue);

    const today = new Date();

    return (
        date.getFullYear() === today.getFullYear() &&
        date.getMonth() === today.getMonth() &&
        date.getDate() === today.getDate()
    );
}


function updateDashboard() {
    const todayFeeding = feedingRecords.filter(
        record => isToday(record.createdAt)
    );

    const todayWater = waterRecords.filter(
        record => isToday(record.createdAt)
    );

    const foodTotal = todayFeeding.reduce(
        (sum, record) => sum + Number(record.quantityGrams || 0),
        0
    );

    const waterOffered = todayWater.reduce(
        (sum, record) => sum + Number(record.offeredMl || 0),
        0
    );

    const waterConsumed = todayWater.reduce(
        (sum, record) =>
            sum + (record.measured ? Number(record.consumedMl || 0) : 0),
        0
    );

    if ($("stat-food")) {
        $("stat-food").textContent = `${foodTotal} g`;
    }

    if ($("stat-feeding-count")) {
        $("stat-feeding-count").textContent = todayFeeding.length;
    }

    if ($("stat-water")) {
        $("stat-water").textContent = `${waterOffered} ml`;
    }

    if ($("stat-water-count")) {
        $("stat-water-count").textContent = todayWater.length;
    }

    if ($("stat-consumed")) {
        $("stat-consumed").textContent = `${waterConsumed} ml`;
    }

    if (catProfile && $("stat-weight")) {
        $("stat-weight").textContent = `${catProfile.weightKg} kg`;
    }

    if ($("dashboard-date")) {
        $("dashboard-date").textContent =
            new Date().toLocaleDateString();
    }

    renderRecentFeeding(todayFeeding);
}


function renderRecentFeeding(records) {
    const container = $("recent-feeding");

    if (!container) return;

    container.innerHTML = "";

    if (records.length === 0) {
        container.textContent = "No feeding activity recorded today.";
        return;
    }

    records.slice(-5).reverse().forEach(record => {
        const item = document.createElement("p");

        item.textContent =
            `${record.foodName} â€” ${record.quantityGrams} g (${record.mealTime})`;

        container.appendChild(item);
    });
}

// ==========================================

async function loadWellness() {
    const date = $("wellness-date")?.value ||
        new Date().toISOString().slice(0, 10);

    try {
        const result = await apiRequest(
            `/api/wellness/daily?date=${encodeURIComponent(date)}`
        );

        const report = result.data || result;

        renderWellness(report);

        return report;
    } catch (error) {
        showMessage(error.message, "error");
        throw error;
    }
}


function renderWellness(report) {
    const container = $("wellness-report");

    if (!container) return;

    if (!report) {
        container.textContent = "No wellness report available.";
        return;
    }

    const feeding = report.feeding || {};
    const water = report.water || {};
    const cat = report.cat || {};

    container.innerHTML = "";

    const details = [
        ["Date", report.date || ""],
        ["Cat", cat.name || "Not available"],
        ["Food records", feeding.totalRecords ?? 0],
        ["Total food", `${feeding.totalGrams ?? 0} g`],
        ["Water records", water.totalRecords ?? 0],
        ["Water offered", `${water.totalOfferedMl ?? 0} ml`],
        [
            "Measured water consumed",
            `${water.totalMeasuredConsumedMl ?? 0} ml`
        ]
    ];

    details.forEach(([label, value]) => {
        const item = document.createElement("p");

        const strong = document.createElement("strong");
        strong.textContent = `${label}: `;

        item.append(strong, document.createTextNode(String(value)));

        container.appendChild(item);
    });
}


if ($("wellness-date")) {
    $("wellness-date").addEventListener("change", loadWellness);
}

async function refreshApp() {
    const status = $("connection-status");

    if (status) {
        status.textContent = "Connecting...";
    }

    try {
        await Promise.all([
            loadCat(),
            loadFeeding(),
            loadWater()
        ]);

        updateDashboard();

        if (status) {
            status.textContent = "Connected";
        }
    } catch (error) {
        if (status) {
            status.textContent = "Connection error";
        }

        console.error("CatCare loading error:", error);
    }
}
